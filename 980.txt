class Solution {
    int[][] grid;
    int rows, cols;
    int toVisit = 0;
    int paths = 0;

    public int uniquePathsIII(int[][] grid) {
        this.grid = grid;
        rows = grid.length;
        cols = grid[0].length;
        int startRow = 0, startCol = 0;

        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                if (grid[i][j] == 0) toVisit++;   // empty cells to walk
                else if (grid[i][j] == 1) {
                    startRow = i;
                    startCol = j;
                }
            }
        }
        toVisit++; // include the ending square itself

        dfs(startRow, startCol, 0);
        return paths;
    }

    private void dfs(int r, int c, int visited) {
        if (r < 0 || r >= rows || c < 0 || c >= cols || grid[r][c] == -1) return;

        if (grid[r][c] == 2) {
            if (visited == toVisit) paths++;
            return;
        }

        int temp = grid[r][c];
        grid[r][c] = -1; // mark visited (temporarily block it)

        dfs(r + 1, c, visited + 1);
        dfs(r - 1, c, visited + 1);
        dfs(r, c + 1, visited + 1);
        dfs(r, c - 1, visited + 1);

        grid[r][c] = temp; // backtrack (unmark)
    }
}
