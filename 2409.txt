class Solution {
    public int countDaysTogether(String arriveAlice, String leaveAlice, String arriveBob, String leaveBob) {
        int [] days = {31, 28, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31};
        int a1 = getDays(arriveAlice ,days);
        int a2 = getDays(leaveAlice ,days);
        int b1 = getDays(arriveBob ,days);
        int b2 = getDays(leaveBob ,days);

        int start = Math.max(a1,b1);
        int end = Math.min(a2,b2);
        if(start > end ) return 0;

        return end-start+1;
        
    }
    public int getDays(String date , int[]days){
        int month = (date.charAt(0)-'0')*10 +(date.charAt(1)-'0');
        int day =  (date.charAt(3)-'0')*10 +(date.charAt(4)-'0');
         int total = day;
         for(int i=0;i<month-1;i++){
            total+=days[i];
         }
         return total;
    }
}
