class Solution {

    Boolean[][] dp;

    public boolean isMatch(String s, String p) {

        dp = new Boolean[s.length() + 1][p.length() + 1];

        return solve(s, p, 0, 0);
    }

    private boolean solve(String s, String p, int i, int j) {

        // Pattern is finished
        if (j == p.length()) {
            return i == s.length();
        }

        // Already calculated
        if (dp[i][j] != null) {
            return dp[i][j];
        }

        // Check current character
        boolean firstMatch =
                i < s.length() &&
                (s.charAt(i) == p.charAt(j)
                || p.charAt(j) == '.');

        // Next character is '*'
        if (j + 1 < p.length() && p.charAt(j + 1) == '*') {

            // '*' matches zero occurrences
            boolean skip = solve(s, p, i, j + 2);

            // '*' matches one or more occurrences
            boolean take = false;

            if (firstMatch) {
                take = solve(s, p, i + 1, j);
            }

            return dp[i][j] = skip || take;
        }

        // Normal character or '.'
        if (firstMatch) {
            return dp[i][j] =
                    solve(s, p, i + 1, j + 1);
        }

        // No match
        return dp[i][j] = false;
    }
}
