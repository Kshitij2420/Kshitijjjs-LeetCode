class Solution {
    public String[] findRelativeRanks(int[] score) {
     int n = score.length;
     PriorityQueue<Integer> maxheap = new PriorityQueue<>((a,b)->score[b]-score[a]);
     for(int i=0;i<n;i++){
        maxheap.add(i);
     }    
     String [] ans = new String[n];
     int rank =1;
     while(!maxheap.isEmpty()){
        int i = maxheap.poll();
        if(rank==1){
            ans[i]="Gold Medal";
        }
        else if(rank==2){
            ans[i]="Silver Medal";
        }
        else if(rank ==3){
            ans[i]="Bronze Medal";
        }
        else{
            ans[i] = String.valueOf(rank);
        }
        rank++;
     }
     return ans;
    }
}
